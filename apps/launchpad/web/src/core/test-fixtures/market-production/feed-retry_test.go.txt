package rest_test

import (
	"encoding/json"
	"fmt"
	"net/http"
	"net/http/httptest"
	"sync"
	"testing"
	"time"

	"github.com/Sidiora-Technologies/KindleLaunch/core/api/internal/rest"
	"github.com/Sidiora-Technologies/KindleLaunch/core/discovery"
)

func TestFeedRetryWebService(t *testing.T) {
	healthy, pool := hsdDevServer(t, true)
	hsdPublishCallout(t, pool, discovery.Callout{ID: hsdDevCallout1, AuthorAccountID: hsdDevWallet, PoolAddress: hsdDevPool, Body: "Public callout", Direction: "bullish", Disclosure: "organic", ModerationStatus: "approved", CreatedAt: time.Now(), Revision: 1})
	unavailable := rest.ResponseMetadata((discovery.HTTP{}).Handler())
	type read struct {
		Status int   `json:"status"`
		At     int64 `json:"at"`
	}
	var mu sync.Mutex
	var reads []read
	failures := 0
	stop := make(chan struct{}, 1)
	mux := http.NewServeMux()
	mux.HandleFunc("POST /reset/{failures}", func(w http.ResponseWriter, r *http.Request) {
		mu.Lock()
		defer mu.Unlock()
		if r.PathValue("failures") == "1" {
			failures = 1
		} else if r.PathValue("failures") == "2" {
			failures = 2
		} else {
			http.Error(w, "Invalid failure count", http.StatusBadRequest)
			return
		}
		reads = nil
		w.WriteHeader(http.StatusNoContent)
	})
	mux.HandleFunc("GET /reads", func(w http.ResponseWriter, r *http.Request) {
		mu.Lock()
		defer mu.Unlock()
		_ = json.NewEncoder(w).Encode(reads)
	})
	mux.HandleFunc("POST /stop", func(w http.ResponseWriter, r *http.Request) {
		w.WriteHeader(http.StatusNoContent)
		stop <- struct{}{}
	})
	mux.HandleFunc("GET /v2/discovery/feed", func(w http.ResponseWriter, r *http.Request) {
		mu.Lock()
		defer mu.Unlock()
		at := time.Now().UnixMilli()
		recorded := httptest.NewRecorder()
		if len(reads) < failures {
			unavailable.ServeHTTP(recorded, r)
			if recorded.Code != http.StatusServiceUnavailable || recorded.Header().Get("Retry-After") != "" {
				t.Errorf("Discovery failure = %d, Retry-After = %q", recorded.Code, recorded.Header().Get("Retry-After"))
			}
		} else {
			healthy.Config.Handler.ServeHTTP(recorded, r)
		}
		reads = append(reads, read{Status: recorded.Code, At: at})
		for key, values := range recorded.Header() {
			w.Header()[key] = values
		}
		w.WriteHeader(recorded.Code)
		_, _ = w.Write(recorded.Body.Bytes())
	})
	server := httptest.NewServer(mux)
	defer server.Close()
	fmt.Printf("FEED_RETRY_SERVICE %s\n", server.URL)
	select {
	case <-stop:
	case <-time.After(time.Minute):
		t.Fatal("Feed session did not finish")
	}
}
