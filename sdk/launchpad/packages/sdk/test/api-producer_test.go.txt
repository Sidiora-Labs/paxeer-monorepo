package rest_test

import (
	"encoding/json"
	"fmt"
	"net/http"
	"net/http/httptest"
	"os"
	"testing"
	"time"

	"github.com/Sidiora-Technologies/KindleLaunch/core/api/internal/broker"
	"github.com/Sidiora-Technologies/KindleLaunch/core/api/internal/rest"
	"github.com/Sidiora-Technologies/KindleLaunch/core/api/internal/sse"
	sharedauth "github.com/Sidiora-Technologies/KindleLaunch/shared/auth"
	"github.com/go-chi/chi/v5"
)

func TestSDKAPIProducer(t *testing.T) {
	_, model, stream, address, _ := productionReadsFixture(t)
	application := os.Getenv("SDK_APPLICATION_ORIGIN")
	stream.Authority = sharedauth.HTTPStreamAuthority{Endpoint: os.Getenv("SDK_IDENTITY_ORIGIN"), ServiceToken: os.Getenv("SDK_IDENTITY_SERVICE_TOKEN")}
	admission, err := rest.NewAdmission(100, 100, 50*time.Millisecond)
	if err != nil {
		t.Fatal(err)
	}
	router := chi.NewRouter()
	if err = rest.RegisterProductionReads(router, rest.ProductionReadOptions{Model: model, Stream: stream, Admission: admission, RegisterSnapshots: true, AllowedOrigins: map[string]bool{application: true}}); err != nil {
		t.Fatal(err)
	}
	hub := sse.NewHub(sse.Options{PrivateStream: stream, AllowedOrigins: []string{application}, Fanout: broker.NewTopicFanout()})
	hub.Register(router)
	server := httptest.NewServer(router)
	defer server.Close()
	stopped := make(chan struct{})
	control := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) { close(stopped); w.WriteHeader(http.StatusNoContent) }))
	defer control.Close()
	data, err := json.Marshal(map[string]any{"kind": "ready", "value": map[string]any{"origin": server.URL, "control": control.URL, "pool": address}})
	if err != nil {
		t.Fatal(err)
	}
	fmt.Printf("SDK_SERVICE %s\n", data)
	select {
	case <-stopped:
	case <-time.After(5 * time.Minute):
		t.Fatal("stream consumer did not finish")
	}
}
