package app

import (
	"encoding/json"
	"fmt"
	"net/http"
	"net/http/httptest"
	"testing"
	"time"

	"github.com/Sidiora-Technologies/KindleLaunch/core/pnl-tracker/internal/production/l18"
	"github.com/go-chi/chi/v5"
)

func TestSDKRewardProducer(t *testing.T) {
	router := chi.NewRouter()
	registerRewardRoutes(router, l18.HTTP{})
	server := httptest.NewServer(router)
	defer server.Close()
	stopped := make(chan struct{})
	control := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) { close(stopped); w.WriteHeader(http.StatusNoContent) }))
	defer control.Close()
	data, err := json.Marshal(map[string]any{"kind": "ready", "value": map[string]any{"origin": server.URL, "control": control.URL}})
	if err != nil {
		t.Fatal(err)
	}
	fmt.Printf("SDK_SERVICE %s\n", data)
	select {
	case <-stopped:
	case <-time.After(time.Minute):
		t.Fatal("reward consumer did not finish")
	}
}
